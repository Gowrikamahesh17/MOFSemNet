# Runtime dependencies for MOFSemNet.
#
# PyTorch: pinned to the CPU wheel by default so `pip install -r requirements.txt` works
# on any machine (personal laptop, university server, Colab CPU runtime) without extra
# index URLs. For a GPU machine, install the matching CUDA wheel *before* this file, e.g.:
#   pip install torch==2.6.0 --index-url https://download.pytorch.org/whl/cu121
# then re-run `pip install -r requirements.txt` — pip will see torch is already satisfied.
# See notebooks/colab_setup.ipynb for the Colab-specific install sequence (torch-geometric's
# compiled extensions need a wheel matched to the exact torch+CUDA combination at runtime).
torch==2.6.0

# PyTorch Geometric — core package only. The optional compiled extensions
# (torch-scatter/torch-sparse/torch-cluster) are NOT required for the GCN/GraphSAGE/GAT
# layers used in this project (torch_geometric.nn.GCNConv/SAGEConv/GATConv run on the
# pure-PyTorch message-passing backend), so they are deliberately left out here to keep
# `pip install` device-agnostic and dependency-free. See notebooks/colab_setup.ipynb if a
# future module needs them.
torch_geometric==2.6.1

# were last verified against. tokenizers/sentencepiece are left unpinned so pip resolves
